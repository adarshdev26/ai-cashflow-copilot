model Transaction {
  id                     String                @id @default(uuid())
  amount                 Float
  type                   TransactionType       @default(DEBIT)
  description            String?
  workspaceId            String
  workspace              Workspace             @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  categoryId             String?
  category               Category?             @relation(fields: [categoryId], references: [id], onDelete: SetNull)
  // Nullable for now; becomes required after the backfill (Commit 13).
  // Restrict: an account with transactions can never be hard-deleted.
  accountId              String?
  account                FinancialAccount?     @relation("TransactionAccount", fields: [accountId], references: [id], onDelete: Restrict)
  // Set only for transactions generated by a schedule.
  recurringTransactionId String?
  recurringTransaction   RecurringTransaction? @relation(fields: [recurringTransactionId], references: [id], onDelete: SetNull)
  occurrenceDate         DateTime?
  createdAt              DateTime              @default(now())
  updatedAt              DateTime              @updatedAt
  toAccountId            String?
  toAccount              FinancialAccount?     @relation("TransactionToAccount", fields: [toAccountId], references: [id], onDelete: Restrict)

  // DB-level idempotency: one transaction per schedule + occurrence.
  // Manual transactions (NULLs) are never affected.
  @@unique([workspaceId, recurringTransactionId, occurrenceDate])
  @@index([workspaceId, createdAt])
  @@index([workspaceId, categoryId])
  @@index([workspaceId, accountId])
  @@index([workspaceId, categoryId, createdAt])
  @@index([recurringTransactionId])
  @@index([workspaceId, toAccountId])

import { NextResponse } from 'next/server';
import { prisma } from '@/lib/db/prisma';
import { processSchedule } from '@/lib/recurring/create-recurring-occurrence';
import { cleanupResetTokens } from '@/lib/auth/cleanup';
import { logAudit } from '@/lib/admin/audit';
import { AUDIT_ACTIONS } from '@/lib/admin/audit-events';

export const dynamic = 'force-dynamic';
export const maxDuration = 60;

async function handle(request: Request) {
    const secret = process.env.CRON_SECRET;
    if (!secret || request.headers.get('authorization') !== `Bearer ${secret}`) {
        return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const now = new Date();
    const due = await prisma.recurringTransaction.findMany({
        where: { isActive: true, nextRunAt: { lte: now } },
        orderBy: { nextRunAt: 'asc' },
        take: 200,
    });

    let generated = 0,
        skipped = 0,
        failed = 0;
    for (const s of due) {
        try {
            const r = await processSchedule(s, now);
            generated += r.generated;
            if (r.skipped) skipped++;
        } catch (e) {
            failed++;
            console.error('recurring failed', s.id, e);
            await prisma.recurringTransaction
                .update({
                    where: { id: s.id },
                    data: { lastError: 'Unexpected error, will retry' },
                })
                .catch((err) => console.error('recurring: failed to persist lastError', s.id, err));
        }
    }

    // Housekeeping runs last and never fails the recurring run.
    let resetTokensDeleted = 0;
    try {
        resetTokensDeleted = await cleanupResetTokens();
    } catch (e) {
        console.error('cron cleanup failed', e);
    }

    if (due.length > 0) {
        await logAudit({
            actor: { type: 'SYSTEM' },
            action: AUDIT_ACTIONS.RECURRING_PROCESSED,
            resourceType: 'recurring',
            metadata: {
                processedCount: due.length,
                generatedCount: generated,
                skippedCount: skipped,
                failedCount: failed,
            },
        });
    }

    return NextResponse.json({
        processed: due.length,
        generated,
        skipped,
        failed,
        resetTokensDeleted,
    });
}

export const GET = handle; // Vercel Cron sends GET
export const POST = handle; // for manual/other schedulers

const COLLAPSED_KEY = 'admin-sidebar-collapsed';
const collapsedListeners = new Set<() => void>();

function subscribeCollapsed(cb: () => void) {
    collapsedListeners.add(cb);
    window.addEventListener('storage', cb);
    return () => {
        collapsedListeners.delete(cb);
        window.removeEventListener('storage', cb);
    };
}
function getCollapsedSnapshot(defaultCollapsed: boolean) {
    try {
        const stored = localStorage.getItem(COLLAPSED_KEY);
        return stored === null ? defaultCollapsed : stored === 'true';
    } catch {
        return defaultCollapsed;
    }
}
function setCollapsedStored(next: boolean) {
    try {
        localStorage.setItem(COLLAPSED_KEY, String(next));
    } catch {}
    collapsedListeners.forEach((listener) => listener());
}

import { NextResponse } from 'next/server';
import type { Prisma } from '@/generated/prisma';
import { prisma } from '@/lib/db/prisma';
import { requireAdmin } from '@/lib/admin/auth';
import { parseListParams, paginationMeta } from '@/lib/admin/list';

const TOKEN_RE = /^[a-z0-9_.-]{1,64}$/i;
const ID_RE = /^[a-zA-Z0-9_-]{1,64}$/;

/** Date-only values (YYYY-MM-DD) are treated as UTC; `dateTo` is inclusive of the whole day. */
function parseDate(raw: string | null, endOfDay: boolean): Date | null | undefined {
    if (!raw) return undefined;
    const dateOnly = /^\d{4}-\d{2}-\d{2}$/.test(raw);
    const d = new Date(dateOnly ? `${raw}T${endOfDay ? '23:59:59.999' : '00:00:00.000'}Z` : raw);
    if (Number.isNaN(d.getTime())) return null;
    if (dateOnly && d.toISOString().slice(0, 10) !== raw) return null;
    return d;
}

export const AUDIT_ROW_SELECT = {
    id: true,
    createdAt: true,
    actorId: true,
    actorType: true,
    actorName: true,
    actorEmail: true,
    action: true,
    resourceType: true,
    resourceId: true,
    workspaceId: true,
} satisfies Prisma.AuditLogSelect;

export async function GET(req: Request) {
    const g = await requireAdmin();
    if (!g.ok) return g.response;

    const url = new URL(req.url);
    const p = parseListParams(
        url,
        ['createdAt', 'action', 'resourceType'],
        'createdAt',
        await getSystemNumber('limits.listMaxPageSize'),
    );

    const sp = url.searchParams;
    const action = sp.get('action');
    const resourceType = sp.get('resourceType');
    const actorId = sp.get('actorId');
    const resourceId = sp.get('resourceId');
    const dateFrom = parseDate(sp.get('dateFrom'), false);
    const dateTo = parseDate(sp.get('dateTo'), true);
    if (dateFrom === null || dateTo === null) {
        return NextResponse.json({ error: 'Invalid date value' }, { status: 400 });
    }
    if (dateFrom && dateTo && dateFrom > dateTo) {
        return NextResponse.json({ error: 'dateFrom must be before dateTo' }, { status: 400 });
    }

    const where: Prisma.AuditLogWhereInput = {
        ...(action && TOKEN_RE.test(action) ? { action } : {}),
        ...(resourceType && TOKEN_RE.test(resourceType) ? { resourceType } : {}),
        ...(actorId && ID_RE.test(actorId) ? { actorId } : {}),
        ...(resourceId && ID_RE.test(resourceId) ? { resourceId } : {}),
        ...(dateFrom || dateTo
            ? {
                  createdAt: {
                      ...(dateFrom ? { gte: dateFrom } : {}),
                      ...(dateTo ? { lte: dateTo } : {}),
                  },
              }
            : {}),
        ...(p.search
            ? {
                  OR: [
                      { actorName: { contains: p.search, mode: 'insensitive' } },
                      { actorEmail: { contains: p.search, mode: 'insensitive' } },
                      { resourceId: { contains: p.search, mode: 'insensitive' } },
                  ],
              }
            : {}),
    };

    const [rows, totalCount] = await Promise.all([
        prisma.auditLog.findMany({
            where,
            select: AUDIT_ROW_SELECT,
            // id is the tie-breaker so pages stay stable when primary values collide
            orderBy: [
                p.sortBy === 'createdAt'
                    ? { createdAt: p.sortDir }
                    : p.sortBy === 'action'
                      ? { action: p.sortDir }
                      : { resourceType: p.sortDir },
                { id: p.sortDir },
            ],
            skip: p.skip,
            take: p.limit,
        }),
        prisma.auditLog.count({ where }),
    ]);

    return NextResponse.json({
        data: rows,
        pagination: paginationMeta(p.page, p.limit, totalCount),
    });
}

export function parseListParams(
    url: URL,
    allowedSort: readonly string[],
    defaultSort: string,
    maxLimit = 100,
) {
    const sp = url.searchParams;
    const page = Math.max(1, Number(sp.get('page')) || 1);
    const limit = Math.min(maxLimit, Math.max(1, Number(sp.get('limit')) || 20));
    const sortRaw = sp.get('sortBy');
    const sortBy = sortRaw && allowedSort.includes(sortRaw) ? sortRaw : defaultSort;
    const sortDir: 'asc' | 'desc' = sp.get('sortDir') === 'asc' ? 'asc' : 'desc';
    const isActiveRaw = sp.get('isActive');
    const isActive = isActiveRaw === 'true' ? true : isActiveRaw === 'false' ? false : undefined;
    const search = sp.get('search')?.trim() || undefined;
    return { page, limit, skip: (page - 1) * limit, sortBy, sortDir, isActive, search };
}

export function paginationMeta(page: number, limit: number, totalCount: number) {
    const totalPages = Math.max(1, Math.ceil(totalCount / limit));
    return { page, limit, totalCount, totalPages, hasMore: page < totalPages };
}

const SENSITIVE_KEY = /pass(word)?|token|secret|hash|authorization|cookie/i;

function redact(value: unknown, depth = 0): unknown {
    if (depth > 5) return '[truncated]';
    if (value instanceof Date) return value.toISOString();
    if (Array.isArray(value)) return value.map((v) => redact(v, depth + 1));
    if (value && typeof value === 'object') {
        return Object.fromEntries(
            Object.entries(value).map(([k, v]) => [
                k,
                SENSITIVE_KEY.test(k) ? '[redacted]' : redact(v, depth + 1),
            ]),
        );
    }
    return value;
}

function getRequestContext(request: Request) {
    const forwarded = request.headers.get('x-forwarded-for')?.split(',')[0]?.trim();
    const ip = forwarded || request.headers.get('x-real-ip') || null;
    const userAgent = request.headers.get('user-agent')?.slice(0, 255) ?? null;
    return { ip, userAgent };
}

/**
 * Best-effort audit write. NEVER throws, so it can't break the main action.
 * Always `await` it in route handlers; un-awaited promises can be dropped on serverless.
 */
export async function logAudit(input: LogAuditInput): Promise<void> {
    try {
        const ctx = input.request
            ? getRequestContext(input.request)
            : { ip: null, userAgent: null };
        await prisma.auditLog.create({
            data: {
                actorId: input.actor.id ?? null,
                actorType: input.actor.type,
                actorName: input.actor.name ?? null,
                actorEmail: input.actor.email ?? null,
                action: input.action,
                resourceType: input.resourceType,
                resourceId: input.resourceId ?? null,
                workspaceId: input.workspaceId ?? null,
                metadata: input.metadata
                    ? (redact(input.metadata) as Prisma.InputJsonValue)
                    : undefined,
                ip: ctx.ip,
                userAgent: ctx.userAgent,
            },
        });
    } catch (err) {
        // Log only the action and error message, never the metadata.
        console.error(
            '[audit] write failed:',
            input.action,
            err instanceof Error ? err.message : 'unknown error',
        );
    }
}

export function getJwtSecret(): string {
    const secret = process.env.JWT_SECRET;
    if (!secret) {
        throw new Error('JWT_SECRET is missing from environment variables.');
    }
    return secret;

// One DB lookup per request, even if getSession() is called several times
// (layout + page + navbar).
const getPasswordChangedAt = cache(async (userId: string) => {
    const user = await prisma.user.findUnique({
        where: { id: userId },
        select: { passwordChangedAt: true, isActive: true },
    });
    return user; // null when the user no longer exists
});

export async function getSessionInfo(): Promise<SessionInfo | null> {
    let decoded: TokenPayload;
    try {
        const cookieStore = await cookies();
        const token = cookieStore.get('token')?.value;
        if (!token) return null;

        decoded = jwt.verify(token, getJwtSecret()) as TokenPayload;
    } catch {
        return null; // missing, expired or tampered
    }

    const user = await getPasswordChangedAt(decoded.userId);
    if (!user) return null;
    if (!user.isActive) return null;

    if (user.passwordChangedAt) {
        const changedAtSec = Math.floor(user.passwordChangedAt.getTime() / 1000);
        if ((decoded.iat ?? 0) < changedAtSec) return null;
    }

    if (decoded.impersonatedBy) {
        const admin = await getAdminSession();
        if (!admin || admin.adminId !== decoded.impersonatedBy) return null;
        return {
            userId: decoded.userId,
            email: decoded.email,
            impersonating: true,
            adminEmail: admin.email,
            impersonatedBy: decoded.impersonatedBy,
        };
    }

    return {
        userId: decoded.userId,
        email: decoded.email,
        impersonating: false,
        adminEmail: null,
        impersonatedBy: null,
    };
}

export async function getSession(): Promise<{ userId: string; email: string } | null> {
    const session = await getSessionInfo();
    if (!session) return null;
    return { userId: session.userId, email: session.email };
}

export async function requireUserId(): Promise<string | null> {
    const session = await getSession();
    return session ? session.userId : null;
}

// ---------------------------------------------------------------------------
// Auth cookie (one source of truth for login, invite register and logout)
// ---------------------------------------------------------------------------

export const AUTH_COOKIE_MAX_AGE_S = 60 * 60 * 24 * 7; // 7 days
export const TOKEN_BACKUP_COOKIE = 'token_backup';

export function signAuthToken(
    user: { id: string; email: string },
    options?: { expiresIn?: string; impersonatedBy?: string; impersonatedAt?: number },
): string {
    const payload = {
        userId: user.id,
        email: user.email,
        ...(options?.impersonatedBy ? { impersonatedBy: options.impersonatedBy } : {}),
        ...(typeof options?.impersonatedAt === 'number'
            ? { impersonatedAt: options.impersonatedAt }
            : {}),
    };

    const signOptions: SignOptions = {
        expiresIn:
            (options?.expiresIn as SignOptions['expiresIn']) ?? ('7d' as SignOptions['expiresIn']),
    };

    return jwt.sign(payload, getJwtSecret(), signOptions);
}

export async function setAuthCookie(
    user: { id: string; email: string },
    options?: {
        expiresInSeconds?: number;
        expiresIn?: string;
        impersonatedBy?: string;
        impersonatedAt?: number;
    },
) {
    const cookieStore = await cookies();
    const existingToken = cookieStore.get('token')?.value;
    const sessionSeconds = (await getSystemNumber('auth.userSessionDays')) * 86400;
    const maxAge = options?.expiresInSeconds ?? sessionSeconds;

    if (options?.impersonatedBy) {
        if (existingToken && !existingToken.includes('impersonatedBy')) {
            cookieStore.set(TOKEN_BACKUP_COOKIE, existingToken, {
                httpOnly: true,
                secure: process.env.NODE_ENV === 'production',
                sameSite: 'strict',
                maxAge: sessionSeconds,
                path: '/',
            });
        }
    }

    cookieStore.set(
        'token',
        signAuthToken(user, {
            expiresIn: options?.expiresIn ?? `${maxAge}s`,
            impersonatedBy: options?.impersonatedBy,
            impersonatedAt: options?.impersonatedAt,
        }),
        {
            httpOnly: true,
            secure: process.env.NODE_ENV === 'production',
            sameSite: 'strict',
            maxAge,
            path: '/',
        },
    );
}

export async function clearAuthCookie() {
    const cookieStore = await cookies();
    cookieStore.set('token', '', {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production',
        sameSite: 'strict',
        maxAge: 0,
        path: '/',
    });
    cookieStore.set(TOKEN_BACKUP_COOKIE, '', {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production',
        sameSite: 'strict',
        maxAge: 0,
        path: '/',
    });
}

/** Renders a single metadata value, choosing the best layout for its shape. */
function MetaValue({ value }: { value: unknown }) {
    if (value === null || value === undefined) {
        return <span className="text-base-content/50">—</span>;
    }
    if (typeof value === 'string' || typeof value === 'number' || typeof value === 'boolean') {
        return <>{String(value)}</>;
    }
    if (Array.isArray(value)) {
        return <JsonBlock value={value} />;
    }
    if (typeof value === 'object') {
        const obj = value as Record<string, unknown>;
        // Detect the { field: { before, after } } changes shape used by update audits.
        if (isChangesMap(obj)) {
            return <ChangesTable changes={obj} />;
        }
        return <JsonBlock value={obj} />;
    }
    return <>{String(value)}</>;
}

/** True when every own value is `{ before?, after? }`. */
function isChangesMap(
    obj: Record<string, unknown>,
): obj is Record<string, { before?: unknown; after?: unknown }> {
    const entries = Object.values(obj);
    if (entries.length === 0) return false;
    return entries.every(
        (v) =>
            v !== null &&
            typeof v === 'object' &&
            !Array.isArray(v) &&
            ('before' in (v as object) || 'after' in (v as object)),
    );
}

function ChangesTable({
    changes,
}: {
    changes: Record<string, { before?: unknown; after?: unknown }>;
}) {
    return (
        <div className="border-base-200 overflow-hidden rounded-md border text-sm">
            <table className="w-full">
                <thead className="bg-base-200/60 text-base-content/60 text-xs">
                    <tr>
                        <th className="px-2 py-1 text-left font-medium">Field</th>
                        <th className="px-2 py-1 text-left font-medium">Before</th>
                        <th className="px-2 py-1 text-left font-medium">After</th>
                    </tr>
                </thead>
                <tbody className="divide-base-200 divide-y">
                    {Object.entries(changes).map(([field, { before, after }]) => (
                        <tr key={field}>
                            <td className="text-base-content/70 px-2 py-1 align-top text-xs">
                                {humanizeKey(field)}
                            </td>
                            <td className="px-2 py-1 align-top">
                                <ChangeCell value={before} variant="before" />
                            </td>
                            <td className="px-2 py-1 align-top">
                                <ChangeCell value={after} variant="after" />
                            </td>
                        </tr>
                    ))}
                </tbody>
            </table>
        </div>
    );
}

function ChangeCell({ value, variant }: { value: unknown; variant: 'before' | 'after' }) {
    if (value === null || value === undefined) {
        return <span className="text-base-content/40 italic">none</span>;
    }
    const text =
        typeof value === 'object'
            ? JSON.stringify(value)
            : typeof value === 'string' && /^[0-9a-f-]{36}$/i.test(value)
              ? // Shorten UUIDs to first 8 chars + ellipsis so rows stay readable.
                `${value.slice(0, 8)}…`
              : String(value);
    const tone = variant === 'before' ? 'text-error/80' : 'text-success/80';
    return (
        <span className={`${tone} font-mono text-xs break-all`} title={String(value)}>
            {text}
        </span>
    );
}

function JsonBlock({ value }: { value: unknown }) {
    return (
        <pre className="bg-base-200 text-base-content overflow-x-auto rounded-md p-2 text-xs">
            {JSON.stringify(value, null, 2)}
        </pre>
    );
}
